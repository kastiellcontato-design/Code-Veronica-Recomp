cmake_minimum_required(VERSION 3.20)
project(VeronicaPKGExtractor VERSION 1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
 message(FATAL_ERROR "This extractor uses Windows CNG and Win32")
endif()
get_filename_component(PORT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/art.rc" "1 ICON \"${PORT}/launcher_art/icon.ico\"\n101 RCDATA \"${PORT}/launcher_art/banner.jpg\"\n1 24 \"${CMAKE_CURRENT_SOURCE_DIR}/app.manifest\"\n")
add_executable(VeronicaPKGExtractor WIN32 app.cpp pkg_core.cpp "${CMAKE_CURRENT_BINARY_DIR}/art.rc")
target_compile_definitions(VeronicaPKGExtractor PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
target_link_libraries(VeronicaPKGExtractor PRIVATE bcrypt gdiplus ole32 shell32 uuid comdlg32 user32 gdi32)
if(MSVC)
 target_link_options(VeronicaPKGExtractor PRIVATE /MANIFEST:NO)
endif()
add_executable(veronica_pkg_tests EXCLUDE_FROM_ALL test_driver.cpp pkg_core.cpp)
target_link_libraries(veronica_pkg_tests PRIVATE bcrypt)
